[计算字符串的镜像分数](https://leetcode.cn/problems/find-mirror-score-of-a-string/)
## 解题思路

[[栈]] + [[哈希表]]：我一开始想用[[哈希表]]直接做，后来发现先前未被标记的镜像值会被覆盖掉，所以还需要借助[[栈]]来存储每一个合法的值，剩下就是模拟

```cpp
class Solution {
public:
    long long calculateScore(string s) {
        long long ans = 0;
        unordered_map<int, vector<int>> p;
        for (int i = 0; i < s.size(); i++) {
            int curr = s[i] - 'a';
            int rev = 25 - curr;
            // 如果镜像值存在
            if (!p[rev].empty()) {
	            // 取一个栈顶值（最近的）
                ans += (i - p[rev].back());
                // 标记（出栈）
                p[rev].pop_back();
            } else {
	            // 将当前值的索引入栈
	            // 这样栈顶存储的就一直是“最近的值”，而不是直接被覆盖掉
                p[curr].push_back(i);
            }
        }
        return ans;
    }
};
```

STL自带容器：

```cpp
class Solution {
public:
    long long calculateScore(string s) {
        long long ans = 0;
        stack<int> p[26];
        for (int i = 0; i < s.size(); i++) {
            int curr = s[i] - 'a';
            if (!p[25 - curr].empty()) {
                ans += (i - p[25 - curr].top());
                p[25 - curr].pop();
            } else {
                p[curr].push(i);
            }
        }
        return ans;
    }
};
```
