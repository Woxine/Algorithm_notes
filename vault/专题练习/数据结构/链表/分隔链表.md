[分隔链表](https://leetcode.cn/problems/split-linked-list-in-parts/)
## 解题思路

[[链表]]：

```cpp
class Solution {
public:
    vector<ListNode*> splitListToParts(ListNode* head, int k) {
        int n = 0;
        ListNode *temp = head;
        // 求得 n
        while (temp) {
            n++;
            temp = temp->next;
        }
        // 求得每个部分的长度
        int length = n / k, remain = n % k;
        ListNode* curr = head;
        vector<ListNode*> parts(k, nullptr);
        for (int i = 0; i < k && curr != nullptr; i++) {
            parts[i] = curr;
            int part_length = length + (i < remain ? 1 : 0);
            for (int j = 1; j < part_length; j++) {
                curr = curr->next;
            }
            ListNode *next = curr->next;
            curr->next = nullptr;
            curr = next;
        }
        return parts;
    }
};
// 链表长度 n
// 每个部分的长度 floor(n/k)
// 用一个额外变量存储 n/k 的余数，加到前几个链表里
```
