[飞地的数量](https://leetcode.cn/problems/number-of-enclaves/)
## 题目拆解 / 题意分析

又是算岛屿面积的，只多了一个环节，就是判断这个岛屿是否在边界，如果在边缘就跳过；否则将面积累加到答案中

## 解题思路

[[网格图]]dfs + [[递归]]：遍历网格，用一个布尔值记录是否在边缘，然后递归计算所有岛屿的面积，更新布尔值，如果不在边缘就累加面积，否则不加

## 代码实现

```cpp
class Solution {
    static constexpr int DIRS[4][2] = {
        {0, -1}, {0, 1}, {-1, 0}, {1, 0}}; // 左右上下
private:
    int dfs(vector<vector<int>>& grid, int i, int j, bool& isleave) {
        int m = grid.size();
        int n = grid[0].size();
        if (i == 0 || i == m - 1 || j == 0 || j == n - 1) {
            isleave = true;
        }
        int cnt = 1;
        grid[i][j] = 0;
        for (auto [dx, dy] : DIRS) {
            int x = i + dx, y = j + dy;
            if (x >= 0 && x < m && y >= 0 && y < n && grid[x][y] == 1) {
                cnt += dfs(grid, x, y, isleave);
            }
        }
        return cnt;
    }

public:
    int numEnclaves(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();
        int ans = 0;

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1) {
                    bool isleave = false;
                    int area = dfs(grid, i, j, isleave);
                    if (!isleave) {
                        ans += area;
                    }
                }
            }
        }
        return ans;
    }
};
```
