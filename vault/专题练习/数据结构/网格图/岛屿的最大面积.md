[岛屿的最大面积](https://leetcode.cn/problems/max-area-of-island/)
## 题目拆解 / 题意分析

跟[[岛屿数量]]其实没什么区别，每次递归返回所有相邻地块的面积和，子问题构成就是岛屿的总面积是当前陆地的面积加上“上下左右”相邻的陆地的面积

## 代码实现

```cpp
class Solution {
    static constexpr int DIRS[4][2] = {
        {0, -1}, {0, 1}, {-1, 0}, {1, 0}}; // 左右上下
public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int ans = 0;
        int row = grid.size();
        int column = grid[0].size();

        auto dfs = [&](this auto&& dfs, int i, int j) -> int {
            int c = 1;  // 岛屿面积
            grid[i][j] = 0;
            for (auto [dx, dy] : DIRS) {
                int x = i + dx, y = j + dy;
                if (x >= 0 && x < row && y >= 0 && y < column && grid[x][y]) {
                    c += dfs(x, y);
                }
            }
            return c;
        };

        for (int i = 0; i < row; i++) {
            for (int j = 0; j < column; j++) {
                if (grid[i][j]) {
                    ans = max(ans, dfs(i, j));
                }
            }
        }
        return ans;
    }
};
```
