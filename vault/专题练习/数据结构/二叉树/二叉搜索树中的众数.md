[二叉搜索树中的众数](https://leetcode.cn/problems/find-mode-in-binary-search-tree/)
## 题目拆解 / 题意分析

## 解题思路

- 要找一堆数中出现次数最多的，很容易想到一个朴素的方法：[[递归]]遍历整棵树，用[[哈希表]]计数，然后输出最大值
- 我们同样能够借助 BST 中序遍历的有序性来做，有序意味着出现的数都是挨在一起的，就能够直接计数而不用存储整个[[二叉树]]了

## 代码实现

```cpp
class Solution {
    unordered_map<int, int> cnt;
    int mx = 0;
    vector<int> ans;

public:
    vector<int> findMode(TreeNode* root) {
        auto dfs = [&](this auto&& dfs, TreeNode* node) -> void {
            if (!node) {
                return;
            }
            dfs(node->left);
            mx = max(mx, ++cnt[node->val]);
            dfs(node->right);
        };
        dfs(root);
        for (auto& [v, c] : cnt) {
            if (c == mx) {
                ans.emplace_back(v);
            }
        }
        return ans;
    }
};
```

```cpp
class Solution {
public:
    int mx, base, cnt = 0;
    vector<int> ans;
    void update(int x) {
        if (x == base) {
            cnt++;
        } else {
            cnt = 1;
            base = x;
        }
        if (cnt == mx) {
            ans.emplace_back(base);
        }
        if (cnt > mx) {
            mx = cnt;
            ans = vector<int>{base};
        }
    }
    vector<int> findMode(TreeNode* root) {
        auto dfs = [&](this auto&& dfs, TreeNode* node) -> void {
            if (!node) {
                return;
            }
            dfs(node->left);
            update(node->val);
            dfs(node->right);
        };
        dfs(root);
        return ans;
    }
};
```
