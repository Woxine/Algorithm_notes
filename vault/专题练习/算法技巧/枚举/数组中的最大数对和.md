[数组中的最大数对和](https://leetcode.cn/problems/max-pair-sum-in-an-array/)
## 解题思路

用[[哈希表]]存储最大数字位对应的最大值，然后不断更新这个最大值以及答案

```cpp
class Solution {
public:
    int max_digit(int n) {
        int ans = 0;
        while (n > 0) {
            ans = max(ans, n % 10);
            n /= 10;
        }
        return ans;
    }
    int maxSum(vector<int>& nums) {
        int ans = -1;
        unordered_map<int, int> map;
        for (int i = 0; i < nums.size(); i++) {
            int biggest = max_digit(nums[i]);
            if (map.find(biggest) != map.end()) {
                ans = max(ans, nums[map[biggest]] + nums[i]);
                if (nums[i] >= nums[map[biggest]]){
                    map[biggest] = i;
                }
                continue;
            }
            map[biggest] = i;
        }
        return ans;
    }
};
```

优化：

```cpp
class Solution {
public:
    int maxSum(vector<int>& nums) {
        int ans = -1;
        // // 直接存储最大数字位对应的最大值，而非索引
        int maxVals[10];
        ranges::fill(maxVals, INT_MIN);
        // 直接遍历值，避免重复使用nums[i]
        for (int n : nums) {
            int biggest = 0;
            for (int i = n; i; i /= 10){
                biggest = max(biggest, i % 10);
            }
            // 更新最大值
            ans = max(ans, maxVals[biggest] + n);
            maxVals[biggest] = max(maxVals[biggest], n);
        }
        return ans;
    }
};
```
## 复杂度分析

### 时间复杂度：**O (n × d)**

其中 n 是数组长度，d 是数字的平均位数

### 空间复杂度：**O (1)**
