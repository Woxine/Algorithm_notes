[爱生气的书店老板](https://leetcode.cn/problems/grumpy-bookstore-owner/description/)

## 解题思路

这道题乍一看多了几个条件，但是本质上还是一道定长[[滑动窗口]]题，只需要将题目要求拆成两个部分，一个是默认条件下有多少顾客满意，再加上老板使用技能最多能再增加多少满意的顾客。关键就在于第二的部分：

```cpp
class Solution {
public:
    int maxSatisfied(vector<int>& customers, vector<int>& grumpy, int minutes) {
        int total = 0;
        int max_window = 0;
        int current_window = 0;
        
        for (int i = 0; i < customers.size(); ++i) {
            if (grumpy[i] == 0) {
	            // 默认条件下有多少顾客满意
                total += customers[i];
            }
            // 使用技能将不满意的顾客变满意
            current_window += grumpy[i] * customers[i];
            if (i >= minutes) {
	            // 离开窗口，恢复使用技能前的状态
                current_window -= grumpy[i - minutes] * customers[i - minutes];
            }
            // 更新能增加的最大顾客数
            max_window = max(max_window, current_window);
        }
        // 默认数量加上最大增加数量
        return total + max_window;
    }
};
```

灵神的题解提供了一种更简洁的解法：

```cpp
class Solution {
public:
    int maxSatisfied(vector<int>& customers, vector<int>& grumpy, int minutes) {
        int s[2]{}, max_s1 = 0;
        for (int i = 0; i < customers.size(); i++) {
            s[grumpy[i]] += customers[i];
            if (i < minutes - 1) { // 窗口长度不足 minutes
                continue;
            }
            max_s1 = max(max_s1, s[1]);
            // 窗口最左边元素离开窗口
            s[1] -= grumpy[i - minutes + 1] ? customers[i - minutes + 1] : 0;
        }
        return s[0] + max_s1;
    }
};
```
