## 解题思路

动态规划：选或不选

## 代码实现

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        target += reduce(nums.begin(), nums.end(), 0);
        if (target < 0 || target % 2) {
            return 0;
        }
        int n = nums.size();
        int m = target / 2;
        vector dp(n + 1, vector<int>(m + 1));
        dp[0][0] = 1;  // 前0个数和为0的情况，是个空集，也算一种
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= m; j++) {
	            // 背包容量不够了，只能不选
                if (nums[i] > j) {
                    dp[i + 1][j] = dp[i][j];
                } else {
                    dp[i + 1][j] = dp[i][j] + dp[i][j - nums[i]];
                }
            }
        }
        return dp[n][m];
    }
};
```

空间优化：每次转移只涉及到两层 i，倒序遍历使得新的状态覆盖掉旧的用不上的状态就行了

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        target += reduce(nums.begin(), nums.end(), 0);
        if (target < 0 || target % 2) {
            return 0;
        }
        int m = target / 2;
        vector<int> dp(m + 1);
        dp[0] = 1; // 前0个数和为0的情况，是个空集，也算一种
        for (int n : nums) {
	        // 倒序遍历，到 n 终止，省去判断容量是否够的过程
            for (int j = m; j >= n; j--) {
                dp[j] += dp[j - n];
            }
        }
        return dp[m];
    }
};
```

### 时间复杂度：**O(nm)**

### 空间复杂度：**O(nm)/O(m)**
