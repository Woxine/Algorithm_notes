## 代码实现

```cpp
class Solution {
public:
    long long maxScore(vector<int>& a, vector<int>& b) {
        int n = b.size();
        vector<vector<long long>> dp(n + 1, vector<long long>(5));
        for (int j = 1; j < 5; j++) {
            dp[0][j] = LLONG_MIN / 2;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 4; j++) {
                dp[i + 1][j + 1] =
                    max(dp[i][j + 1], dp[i][j] + 1LL * a[j] * b[i]);
            }
        }
        return dp[n][4];
    }
};
```

空间优化：

```cpp
class Solution {
public:
    long long maxScore(vector<int>& a, vector<int>& b) {
        int n = b.size();
        long long dp[5]{};
        fill(dp + 1, dp + 5, LLONG_MIN / 2);
        for (int i = 0; i < n; i++) {
	        // 倒序遍历
            for (int j = 3; j >= 0; j--) {
                dp[j + 1] = max(dp[j + 1], dp[j] + 1LL * a[j] * b[i]);
            }
        }
        return dp[4];
    }
};
```
