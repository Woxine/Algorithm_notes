## 代码实现

```cpp
class Solution {
public:
    int numDistinct(string s, string t) {
        int n = s.size();
        int m = t.size();
        if (n < m) {
            return 0;
        }
        vector dp(n + 1, vector<unsigned>(m + 1));
        for (int i = 0; i < n; i++) {
            dp[i][0] = 1;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dp[i + 1][j + 1] =
                    s[i] == t[j] ? dp[i][j + 1] + dp[i][j] : dp[i][j + 1];
            }
        }
        return dp[n][m];
    }
};
```

```cpp
class Solution {
public:
    int numDistinct(string s, string t) {
        int n = s.size();
        int m = t.size();
        if (n < m) {
            return 0;
        }
        vector<unsigned> dp(m + 1);
        dp[0] = 1;
        for (char& c : s) {
            for (int j = m - 1; j >= 0; j--) {
                dp[j + 1] = c == t[j] ? dp[j + 1] + dp[j] : dp[j + 1];
            }
        }
        return dp[m];
    }
};
```
