[72. 编辑距离 - 力扣（LeetCode）](https://leetcode.cn/problems/edit-distance/description/)
## 题目拆解 / 题意分析

## 解题思路

[[动态规划]]：
- 当前操作：可以对一个单词的每个字符进行三种操作：插入，删除，替换（或者什么都不做）
- 状态定义：定义`dp[i][j]`表示将 word1 的前 i 个字符 变成 word 2 的前 j 个字符的最小操作数
- 转移方程：`dp[i + 1][j + 1] = min(dp[i][j], min(dp[i + 1][j], dp[i][j + 1])) + 1;`
## 代码实现

```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size();
        int m = word2.size();
        vector dp(n + 1, vector<int>(m + 1));
        // dfs(0,j) 非法，需要初始化
        for (int j = 0; j < m; j++) {
            dp[0][j + 1] = j + 1;
        }
        for (int i = 0; i < n; i++) {
            dp[i + 1][0] = i + 1; // 初始化 dfs(i,0)
            for (int j = 0; j < m; j++) {
                if (word1[i] == word2[j]) {
                    dp[i + 1][j + 1] = dp[i][j];
                } else {
                    dp[i + 1][j + 1] =
                        min(dp[i][j], min(dp[i + 1][j], dp[i][j + 1])) + 1;
                }
            }
        }
        return dp[n][m];
    }
};
```

空间优化：原理与上一题类似

```cpp
class Solution {
public:
    int minDistance(string s1, string s2) {
        int n = s1.size();
        int m = s2.size();
        vector<int> dp(m + 1);
        for (int j = 0; j < m; j++) {
            dp[j + 1] = j + 1;
        }
        for (int i = 0; i < n; i++) {
            int pre = dp[0]++;
            for (int j = 0; j < m; j++) {
                int temp = dp[j + 1];
                dp[j + 1] = s1[i] == s2[j] ? 
                            pre : 
                            min({pre, dp[j], dp[j + 1]}) + 1;
                pre = temp;
            }
        }
        return dp[m];
    }
};
```
