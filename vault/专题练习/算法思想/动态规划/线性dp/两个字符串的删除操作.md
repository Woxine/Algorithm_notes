[583. 两个字符串的删除操作 - 力扣（LeetCode）](https://leetcode.cn/problems/delete-operation-for-two-strings/description/)
## 题目拆解 / 题意分析

要想删除的字符最少，就是要剩下的[[最长公共子序列]]最大，于是就解决了

## 解题思路

[[动态规划]]：用总长度减去求出来的长度

## 代码实现

```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size();
        int m = word2.size();
        vector dp(n + 1, vector<int>(m + 1));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dp[i + 1][j + 1] = word1[i] == word2[j]
                                       ? dp[i][j] + 1
                                       : max(dp[i][j + 1], dp[i + 1][j]);
            }
        }
        int lcs = dp[n][m];
        return m + n - 2 * lcs;
    }
};
```

空间优化：

```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size();
        int m = word2.size();
        vector<int> dp(m + 1);
        for (char& x : word1) {
            for (int j = 0, pre = 0; j < m; j++) {
                int temp = dp[j + 1];
                dp[j + 1] = x == word2[j] ? pre + 1 : max(dp[j + 1], dp[j]);
                pre = temp;
            }
        }
        int lcs = dp[m];
        return m + n - 2 * lcs;
    }
};
```
