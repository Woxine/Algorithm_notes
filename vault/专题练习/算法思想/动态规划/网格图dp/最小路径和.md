[最小路径和](https://leetcode.cn/problems/minimum-path-sum/)
## 题目拆解 / 题意分析

"只能向下或向右走"意味着问题只会越来越小，最终汇集到一个点上；我们可以从终点开始思考，要想到达最右下角，要么从它的上面来，要么从它的左边来；当然对于网格中的所有点都是一样的，于是我们就得到了状态和状态转移方程：

- **`dp[i][j]`**：表示从网格左上角 `(0,0)` 到达位置 `(i-1, j-1)` 的最小路径和
- 使用 `(m+1) × (n+1)` 的DP数组是为了处理边界情况更方便
- 初始化为 `INT_MAX` 表示初始时所有位置都不可达
- 状态转移方程：
```cpp
dp[i + 1][j + 1] = min(dp[i][j + 1], dp[i + 1][j]) + grid[i][j];
```

## 代码实现

```cpp
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();
        // 初始值全设为最大值，使得 min 函数中只会取到合法路径
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, INT_MAX));
        dp[1][1] = grid[0][0];  // 初始化第一个格子
        
        // 此前的 dp[1][1] = grid[0][0] 和 vector<int>(n + 1, INT_MAX)
        // 使得第一行和第一列的值只会从左边和上面得到
        // 如果不这么做，也可以单独写两个循环进行初始化
        // 那么这两个初始化会是线性的
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (j == 0 && i == 0) {
                    continue;
                }
                int x = grid[i][j];
                dp[i + 1][j + 1] = min(dp[i][j + 1], dp[i + 1][j]) + x;
            }
        }
        return dp[m][n];
    }
};
```

更易读的版本：

```cpp
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size();
        int n = grid[0].size();
        // 由于不需要处理边界条件，也就不需要多开一层了
        vector dp(m, vector<int>(n));
        // 递推起点
        dp[0][0] = grid[0][0];
        // 初始化过程，对第一列和第一行单独处理
        for (int j = 1; j < n; ++j) {
            dp[0][j] = dp[0][j - 1] + grid[0][j];
        }
        for (int i = 1; i < m; ++i) {
            dp[i][0] = dp[i - 1][0] + grid[i][0];
        }
        // 正式开始递推
        for (int i = 1; i < m; i++) {
            for (int j = 1; j < n; j++) {
                dp[i][j] = min(dp[i][j - 1], dp[i - 1][j]) + grid[i][j];
            }
        }
        return dp[m - 1][n - 1];
    }
};
```

还有高手：

```cpp
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int n = grid[0].size();
        vector dp(n + 1, INT_MAX);
        dp[1] = 0;
        for (auto& row : grid) {
            for (int j = 0; j < n; j++) {
                dp[j + 1] = min(dp[j + 1], dp[j]) + row[j];
            }
        }
        return dp[n];
    }
};
```
