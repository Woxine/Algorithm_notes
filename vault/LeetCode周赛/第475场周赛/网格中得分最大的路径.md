[3742. 网格中得分最大的路径 - 力扣（LeetCode）](https://leetcode.cn/problems/maximum-path-score-in-a-grid/)
## 题目拆解 / 题意分析

一看就是标准的[[网格图]]dp，只能向右或向下移动，那么就不会出现环状结构，递归过程中结构只会越来越简单

## 解题思路

- 一般[[递归]]遍历，类似二叉树：递归边界为出界或超出预算，如果到达终点直接返回得分；所以递归过程中需要三个变量分别存储当前的得分，花费和剩下的预算；同时需要一个记忆数组存储已经计算过的路径，减少计算量
- [[递归]] + dfs + [[动态规划]]：

## 代码实现

- 比赛时候的代码，普通递归，非常慢，但是过了

```cpp
class Solution {
public:
    int maxPathScore(vector<vector<int>>& grid, int k) {
        int row = grid.size(), column = grid[0].size();
        vector<vector<vector<int>>> memo(
            row, vector<vector<int>>(column, vector<int>(k + 1, -2)));

        auto dfs = [&](this auto&& dfs, int i, int j, int remain) -> int {
            // 递归边界
            if (i >= row || j >= column) {
                return -1;
            }
            //超预算就返回-1
            int curCost = grid[i][j] != 0;
            if (curCost > remain) {
                return -1;
            }
            // 计算当前分数，抵达终点就返回分数
            int curScore = grid[i][j];
            if (i == row - 1 && j == column - 1) {
                return curScore;
            }
            // 记忆化搜索，如果计算过就直接返回
            if (memo[i][j][remain] != -2) {
                return memo[i][j][remain];
            }
            // 递归遍历右边和下面
            int right = dfs(i, j + 1, remain - curCost);
            int down = dfs(i + 1, j, remain - curCost);
            int maxScore = -1;
            // 如果没超预算，更新最大值
            if (right != -1) {
                maxScore = max(maxScore, curScore + right);
            }
            if (down != -1) {
                maxScore = max(maxScore, curScore + down);
            }
            // 记录有效值
            memo[i][j][remain] = maxScore;
            return maxScore;
        };
        return dfs(0, 0, k);
    }
};
```

- [[动态规划]] + 递归：

```cpp
class Solution {
public:
    int maxPathScore(vector<vector<int>>& grid, int k) {
        int row = grid.size(), column = grid[0].size();
        k = min(k, row + column - 2);
        vector<vector<vector<int>>> memo(
            row, vector<vector<int>>(column, vector<int>(k + 1, -1)));

        auto dfs = [&](this auto&& dfs, int i, int j, int remain) -> int {
            // 递归边界(出界或超预算)
            if (i < 0 || j < 0 || remain < 0) {
                return INT_MIN;
            }
            // 递归终点（网格图起点）
            if (i == 0 && j == 0) {
                return 0;
            }
            // 记忆化搜索，如果计算过就直接返回
            int& res = memo[i][j][remain];
            if (res != -1) {
                return res;
            }
            int curScore = grid[i][j];
            remain -= curScore > 0;
            return res = max(dfs(i - 1, j, remain), dfs(i, j - 1, remain)) +
                         curScore;
        };
        int ans = dfs(row - 1, column - 1, k);
        return ans < 0 ? -1 : ans;
    }
};
```

- 动态规划 + 递推 + 记忆化搜索（灵神题解）：

```cpp
class Solution {
public:
    int maxPathScore(vector<vector<int>>& grid, int K) {
        int m = grid.size(), n = grid[0].size();
        vector f(m + 1, vector(n + 1, vector<int>(K + 2, INT_MIN)));
        ranges::fill(f[0][1].begin() + 1, f[0][1].end(), 0);

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                int x = grid[i][j];
                for (int k = 0; k <= K; k++) {
                    int new_k = k - (x > 0);
                    f[i + 1][j + 1][k + 1] = 
                    max(f[i][j + 1][new_k + 1], f[i + 1][j][new_k + 1]) + x;
                }
            }
        }

        int ans = f[m][n][K + 1];
        return ans < 0 ? -1 : ans;
    }
};
```

### 时间复杂度：**O(mnk)**

其中 m = row，n = column，最坏的情况下O(mn(m+n))

### 空间复杂度：**O(mnk)**
