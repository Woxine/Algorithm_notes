```cpp
class Solution {
public:
    vector<int> minOperations(vector<int>& nums) {
        vector<int> res;
        for (int x : nums) {
            int d1 = 1e9, d2 = 1e9;
            for (int i = x;; i++) {
                if (isPal(i)) {
                    d1 = i - x;
                    break;
                }
            }
            for (int i = x;; i--) {
                if (isPal(i)) {
                    d2 = x - i;
                    break;
                }
            }
            res.push_back(min(d1, d2));
        }
        return res;
    }

private:
    bool isPal(int x) {
        string s = format("{:b}", x);
        for (int i = 0, j = s.size() - 1; i < j; i++, j--) {
            if (s[i] != s[j]) {
                return false;
            }
        }
        return true;
    }
};
```
