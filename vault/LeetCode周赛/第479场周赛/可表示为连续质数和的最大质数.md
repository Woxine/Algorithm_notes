[3770. 可表示为连续质数和的最大质数 - 力扣（LeetCode）](https://leetcode.cn/problems/largest-prime-from-consecutive-prime-sum/description/)
## 题目拆解 / 题意分析

题目本身的很直白，只需要从2开始累加质数，然后判断结果是否小于 n 并且是个质数就行了

但是暴力法O($n\sqrt{n}$)会超时，而埃氏筛O(nlog(logn))不会

## 解题思路

埃氏筛 + 枚举：

## 代码实现

```cpp
class Solution {
public:
	// 埃氏筛
    void init(int n, vector<bool>& is_prime) {
        is_prime[0] = is_prime[1] = false;
        for (int i = 2; i * i <= n; i++) {
            if (is_prime[i]) {
                for (int j = i * i; j <= n; j += i) {
                    is_prime[j] = false;
                }
            }
        }
    }
    int largestPrime(int n) {
        if (n < 2)
            return 0;
        vector<bool> is_prime(n + 1, true);
        init(n,is_prime);
        long long ans = 0;
        long long sum = 0LL;
        // 遍历所有质数
        for (int i = 2; i <= n; i++) {
            if (is_prime[i]) {
                sum += i;
                if (sum > n)
                    break;
                if (is_prime[sum])
                    ans = sum;
            }
        }
        return ans;
    }
};
```

// 2 ~ n 是要找的质数的范围
vector<bool> is_prime(n + 1, true);
void init(int n, vector<bool>& is_prime) {
	// 0 和 1 不是质数
	is_prime[0] = is_prime[1] = false;
	for (int i = 2; i * i <= n; i++) {
		if (is_prime[i]) {
			// 批量处理素数的倍数
			for (int j = i * i; j <= n; j += i) {
				is_prime[j] = false;
			}
		}
	}
}
