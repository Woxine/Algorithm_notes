## 代码实现

原：

```cpp
class Solution {
public:
    int minimumFlips(int n) {
        bitset<32> b(n);                  // 相当于二进制数组
        string s = b.to_string();         // 得到二进制字符串
        auto os = s.substr(s.find('1'));  // 去掉前导零
        reverse(s.begin(), s.end());      // 得到32位反转后的字符串
        int ans = 0;
        // 长度以os为准，消除s反转后一串0在后面的影响
        for (int i = 0; i < os.size(); i++) {
            if (os[i] != s[i]) {
                ans++;
            }
        }
        return ans;
    }
};
```

题解：

```cpp
class Solution {
public:
    int minimumFlips(int n) {
        string s = format("{:b}",n);
        int ans = 0, m = s.size();
        for (int i = 0; i < m / 2; i++) {
	        // 相当于一次比了两位
            if (s[i] != s[m - 1 - i]) {
                ans += 2;
            }
        }
        return ans;
    }
};
```
